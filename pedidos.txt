type EstadoPedido = "pendiente" | "en_impresion" | "listo" | "entregado";
type EstadoPago = "sin_pagar" | "anticipo" | "pagado";
type EnvioTipo = "recogida" | "domicilio" | "transporte";

interface ChecklistItem {
  id: string;
  label: string;
  hecho: boolean;
}

interface EventoPedido {
  id: string;
  texto: string;
  fecha: string;
}

interface PedidoUI {
  id: string;
  codigo: string;
  cliente: { nombre: string; telefono: string; direccion?: string };
  piezaDescripcion: string;
  estado: EstadoPedido;
  fechaEstimadaListo: string;
  horasImpresionEstimadas: number;
  urgente: boolean;
  pago: {
    estado: EstadoPago;
    total: number;
    anticipoPct: number;
    montoCobrado: number;
    metodo: "efectivo" | "qr";
    verificado: boolean;
  };
  envio: {
    tipo: EnvioTipo;
    costo: number;
    tracking?: string;
  };
  impresion?: {
    impresora: string;
    filamento: string;
    gramosPlanificados: number;
    horasPlanificadas: number;
    resultado: "en_progreso" | "pendiente" | "exitoso" | "fallo";
  };
  checklist: ChecklistItem[];
  eventos: EventoPedido[];
}

interface PiezaTopUI {
  id: string;
  nombre: string;
  vecesCotizada: number;
  precioPromedio: number;
  ultimaVez: string;
}

function KpiChip({ icon, label, value, color }: { icon: keyof typeof Ionicons.glyphMap; label: string; value: string; color: string }) {
  return (
    <View style={styles.kpiChip}>
      <View style={[styles.kpiIconWrap, { backgroundColor: color + "22" }]}>
        <Ionicons name={icon} size={16} color={color} />
      </View>
      <View>
        <Text style={styles.kpiValue}>{value}</Text>
        <Text style={styles.kpiLabel}>{label}</Text>
      </View>
    </View>
  );
}

function PiezaTopChip({ pieza }: { pieza: PiezaTopUI }) {
  return (
    <View style={styles.piezaChip}>
      <View style={{ flex: 1 }}>
        <Text style={styles.piezaNombre} numberOfLines={1}>{pieza.nombre}</Text>
        <Text style={styles.piezaMeta}>{pieza.vecesCotizada} veces · {fmtMoneda(pieza.precioPromedio)} prom.</Text>
      </View>
      <Pressable style={styles.reCotizarBtn}>
        <Ionicons name="refresh" size={13} color="#fff" />
      </Pressable>
    </View>
  );
}

function PedidoCard({ pedido, onPress }: { pedido: PedidoUI; onPress: () => void }) {
  const cfg = ESTADO_CFG[pedido.estado];
  const pagoCfg = PAGO_CFG[pedido.pago.estado];
  const checklistDone = pedido.checklist.filter((c) => c.hecho).length;

  return (
    <Pressable style={styles.pedidoCard} onPress={onPress}>
      <View style={styles.sectionHeaderRow}>
        <View style={styles.pedidoCodigoRow}>
          <Text style={styles.pedidoCodigo}>{pedido.codigo}</Text>
          {pedido.urgente && (
            <View style={styles.urgenteBadge}>
              <Ionicons name="flame" size={11} color={theme.danger} />
              <Text style={styles.urgenteBadgeText}>Urgente</Text>
            </View>
          )}
        </View>
        <View style={[styles.estadoPill, { backgroundColor: cfg.bg }]}>
          <Ionicons name={cfg.icon} size={12} color={cfg.color} />
          <Text style={[styles.estadoPillText, { color: cfg.color }]}>{cfg.label}</Text>
        </View>
      </View>

      <Text style={styles.pedidoCliente}>{pedido.cliente.nombre}</Text>
      <Text style={styles.pedidoPieza} numberOfLines={1}>{pedido.piezaDescripcion}</Text>

      <View style={styles.pedidoFootRow}>
        <View style={[styles.pagoPill, { backgroundColor: pagoCfg.bg }]}>
          <Text style={[styles.pagoPillText, { color: pagoCfg.color }]}>{pagoCfg.label}</Text>
        </View>
        <View style={styles.pedidoFootItem}>
          <Ionicons name="cube-outline" size={12} color={theme.textMuted} />
          <Text style={styles.pedidoFootText}>
            {ENVIO_LABEL[pedido.envio.tipo]}
          </Text>
        </View>
        <View style={styles.pedidoFootItem}>
          <Ionicons name="checkbox-outline" size={12} color={theme.textMuted} />
          <Text style={styles.pedidoFootText}>{checklistDone}/{pedido.checklist.length}</Text>
        </View>
      </View>

      <View style={styles.pedidoTimeRow}>
        <Ionicons name="time-outline" size={12} color={theme.textMuted} />
        <Text style={styles.pedidoTimeText}>{pedido.fechaEstimadaListo}</Text>
        <Text style={styles.pedidoMontoText}>{fmtMoneda(pedido.pago.total)}</Text>
      </View>
    </Pressable>
  );
}

function DetallePedidoModal({ pedido, onClose }: { pedido: PedidoUI | null; onClose: () => void }) {
  if (!pedido) return null;
  const cfg = ESTADO_CFG[pedido.estado];
  const pagoCfg = PAGO_CFG[pedido.pago.estado];
  const saldo = pedido.pago.total - pedido.pago.montoCobrado;

  return (
    <Modal visible={!!pedido} animationType="slide" transparent onRequestClose={onClose}>
      <View style={styles.modalBackdrop}>
        <View style={styles.modalSheet}>
          <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={{ paddingBottom: 24 }}>
            <View style={styles.modalHandle} />

            <View style={styles.sectionHeaderRow}>
              <Text style={styles.modalTitle}>{pedido.codigo}</Text>
              <Pressable onPress={onClose} hitSlop={10}>
                <Ionicons name="close" size={22} color={theme.textMuted} />
              </Pressable>
            </View>

            <View style={[styles.estadoPill, { backgroundColor: cfg.bg, alignSelf: "flex-start" }]}>
              <Ionicons name={cfg.icon} size={13} color={cfg.color} />
              <Text style={[styles.estadoPillText, { color: cfg.color }]}>{cfg.label}</Text>
            </View>

            {/* Cliente */}
            <View style={styles.modalBlock}>
              <Text style={styles.modalBlockTitle}>Cliente</Text>
              <Text style={styles.modalLine}>{pedido.cliente.nombre}</Text>
              <Text style={styles.modalLineMuted}>{pedido.cliente.telefono}</Text>
              {pedido.cliente.direccion ? <Text style={styles.modalLineMuted}>{pedido.cliente.direccion}</Text> : null}
              <View style={{ flexDirection: "row", gap: 10, marginTop: 10 }}>
                <Pressable style={styles.contactBtn}>
                  <Ionicons name="call-outline" size={14} color={theme.text} />
                  <Text style={styles.contactBtnText}>Llamar</Text>
                </Pressable>
                <Pressable style={[styles.contactBtn, { backgroundColor: theme.successBg }]}>
                  <Ionicons name="logo-whatsapp" size={14} color={theme.success} />
                  <Text style={[styles.contactBtnText, { color: theme.success }]}>WhatsApp</Text>
                </Pressable>
              </View>
            </View>

            {/* Pieza */}
            <View style={styles.modalBlock}>
              <Text style={styles.modalBlockTitle}>Pieza</Text>
              <Text style={styles.modalLine}>{pedido.piezaDescripcion}</Text>
              {pedido.impresion ? (
                <>
                  <Text style={styles.modalLineMuted}>
                    {pedido.impresion.impresora} · {pedido.impresion.filamento}
                  </Text>
                  <Text style={styles.modalLineMuted}>
                    {pedido.impresion.gramosPlanificados}g planificados · {pedido.impresion.horasPlanificadas}h de impresión
                  </Text>
                </>
              ) : null}
            </View>

            {/* Pago */}
            <View style={styles.modalBlock}>
              <View style={styles.sectionHeaderRow}>
                <Text style={styles.modalBlockTitle}>Pago</Text>
                <View style={[styles.pagoPill, { backgroundColor: pagoCfg.bg }]}>
                  <Text style={[styles.pagoPillText, { color: pagoCfg.color }]}>{pagoCfg.label}</Text>
                </View>
              </View>
              <View style={styles.pagoGrid}>
                <View>
                  <Text style={styles.modalLineMuted}>Total</Text>
                  <Text style={styles.modalLine}>{fmtMoneda(pedido.pago.total)}</Text>
                </View>
                <View>
                  <Text style={styles.modalLineMuted}>Cobrado</Text>
                  <Text style={styles.modalLine}>{fmtMoneda(pedido.pago.montoCobrado)}</Text>
                </View>
                <View>
                  <Text style={styles.modalLineMuted}>Saldo</Text>
                  <Text style={[styles.modalLine, saldo > 0 && { color: theme.warning }]}>{fmtMoneda(saldo)}</Text>
                </View>
              </View>
              <Text style={styles.modalLineMuted}>
                Método: {pedido.pago.metodo.toUpperCase()} · {pedido.pago.verificado ? "Comprobante verificado" : "Pendiente de verificar"}
              </Text>
              {pedido.pago.estado !== "pagado" && (
                <Pressable style={styles.primaryBtn}>
                  <Text style={styles.primaryBtnText}>
                    {pedido.pago.estado === "sin_pagar" ? "Verificar pago y pasar a producción" : "Confirmar pago y entrega"}
                  </Text>
                </Pressable>
              )}
            </View>

            {/* Envío */}
            <View style={styles.modalBlock}>
              <Text style={styles.modalBlockTitle}>Envío</Text>
              <Text style={styles.modalLine}>{ENVIO_LABEL[pedido.envio.tipo]}</Text>
              {pedido.envio.costo > 0 && <Text style={styles.modalLineMuted}>Costo: {fmtMoneda(pedido.envio.costo)}</Text>}
              {pedido.envio.tracking && <Text style={styles.modalLineMuted}>Tracking: {pedido.envio.tracking}</Text>}
            </View>

            {/* Checklist */}
            <View style={styles.modalBlock}>
              <Text style={styles.modalBlockTitle}>Checklist de entrega</Text>
              {pedido.checklist.map((item) => (
                <Pressable key={item.id} style={styles.checklistRow}>
                  <Ionicons
                    name={item.hecho ? "checkbox" : "square-outline"}
                    size={18}
                    color={item.hecho ? theme.success : theme.textMuted}
                  />
                  <Text style={[styles.checklistText, item.hecho && { color: theme.textMuted, textDecorationLine: "line-through" }]}>
                    {item.label}
                  </Text>
                </Pressable>
              ))}
            </View>

            {/* Timeline */}
            <View style={styles.modalBlock}>
              <Text style={styles.modalBlockTitle}>Historial</Text>
              {pedido.eventos.map((ev) => (
                <View key={ev.id} style={styles.eventoRow}>
                  <View style={styles.eventoDot} />
                  <View style={{ flex: 1 }}>
                    <Text style={styles.modalLine}>{ev.texto}</Text>
                    <Text style={styles.modalLineMuted}>{ev.fecha}</Text>
                  </View>
                </View>
              ))}
            </View>
          </ScrollView>
        </View>
      </View>
    </Modal>
  );
}

// ────────────────────────────────────────────────────────────────────────────
// SCREEN
// ────────────────────────────────────────────────────────────────────────────

export default function PedidosScreen() {
  const [filtro, setFiltro] = useState<"todos" | EstadoPedido>("todos");
  const [busqueda, setBusqueda] = useState("");
  const [pedidoActivo, setPedidoActivo] = useState<PedidoUI | null>(null);

  const conteos = useMemo(() => {
    const base: Record<string, number> = { todos: MOCK_PEDIDOS.length };
    (["pendiente", "en_impresion", "listo", "entregado"] as EstadoPedido[]).forEach((k) => {
      base[k] = MOCK_PEDIDOS.filter((p) => p.estado === k).length;
    });
    return base;
  }, []);

  const kpis = useMemo(() => {
    const porCobrar = MOCK_PEDIDOS.reduce((acc, p) => acc + (p.pago.total - p.pago.montoCobrado), 0);
    const urgentes = MOCK_PEDIDOS.filter((p) => p.urgente).length;
    const activos = MOCK_PEDIDOS.filter((p) => p.estado !== "entregado").length;
    const listos = MOCK_PEDIDOS.filter((p) => p.estado === "listo").length;
    return { porCobrar, urgentes, activos, listos };
  }, []);

  const pedidosFiltrados = useMemo(() => {
    return MOCK_PEDIDOS.filter((p) => {
      const pasaFiltro = filtro === "todos" || p.estado === filtro;
      const pasaBusqueda =
        busqueda.trim().length === 0 ||
        p.codigo.toLowerCase().includes(busqueda.toLowerCase()) ||
        p.cliente.nombre.toLowerCase().includes(busqueda.toLowerCase());
      return pasaFiltro && pasaBusqueda;
    });
  }, [filtro, busqueda]);

  return (
    <SafeAreaView style={styles.safe} edges={["top"]}>
      <FlatList
        data={pedidosFiltrados}
        keyExtractor={(item) => item.id}
        contentContainerStyle={styles.listContent}
        showsVerticalScrollIndicator={false}
        renderItem={({ item }) => <PedidoCard pedido={item} onPress={() => setPedidoActivo(item)} />}
        ListEmptyComponent={
          <View style={styles.emptyState}>
            <Ionicons name="file-tray-outline" size={30} color={theme.textMuted} />
            <Text style={styles.emptyTitle}>No hay pedidos en este filtro</Text>
          </View>
        }
        ListHeaderComponent={
          <View style={{ gap: 18, marginBottom: 6 }}>
            <Text style={styles.screenTitle}>Pedidos</Text>

            {/* KPIs */}
            <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={{ gap: 10, paddingRight: 16 }}>
              <KpiChip icon="albums-outline" label="Activos" value={`${kpis.activos}`} color={theme.primary} />
              <KpiChip icon="cash-outline" label="Por cobrar" value={fmtMoneda(kpis.porCobrar)} color={theme.warning} />
              <KpiChip icon="flame-outline" label="Urgentes" value={`${kpis.urgentes}`} color={theme.danger} />
              <KpiChip icon="checkmark-circle-outline" label="Listos" value={`${kpis.listos}`} color={theme.success} />
            </ScrollView>

            {/* Piezas más cotizadas */}
            <View>
              <View style={styles.sectionHeaderRow}>
                <Text style={styles.sectionTitle}>Piezas más cotizadas</Text>
                <Text style={styles.sectionAction}>Ver todas</Text>
              </View>
              <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={{ gap: 10, paddingRight: 16 }}>
                {MOCK_PIEZAS_TOP.map((p) => (
                  <PiezaTopChip key={p.id} pieza={p} />
                ))}
              </ScrollView>
            </View>

            {/* Buscador */}
            <View style={styles.searchBox}>
              <Ionicons name="search" size={16} color={theme.textMuted} />
              <TextInput
                value={busqueda}
                onChangeText={setBusqueda}
                placeholder="Buscar por código o cliente"
                placeholderTextColor={theme.textMuted}
                style={styles.searchInput}
              />
            </View>

            {/* Filtros */}
            <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={{ gap: 8, paddingRight: 16 }}>
              {FILTROS.map((f) => {
                const activo = filtro === f.key;
                return (
                  <Pressable
                    key={f.key}
                    onPress={() => setFiltro(f.key)}
                    style={[styles.filtroChip, activo && { backgroundColor: theme.primary, borderColor: theme.primary }]}
                  >
                    <Text style={[styles.filtroChipText, activo && { color: "#fff" }]}>
                      {f.label} · {conteos[f.key] ?? 0}
                    </Text>
                  </Pressable>
                );
              })}
            </ScrollView>
          </View>
        }
      />

      <DetallePedidoModal pedido={pedidoActivo} onClose={() => setPedidoActivo(null)} />
    </SafeAreaView>
  );
}
