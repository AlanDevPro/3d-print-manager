  `Bs ${n.toLocaleString("es-BO", { minimumFractionDigits: 0 })}`;

    color: theme.success,

  mantenimiento: {

function SectionHeader({
  title,
  subtitle,
  actionLabel,
  onAction,
}: {
  title: string;
  subtitle?: string;
  actionLabel?: string;
  onAction?: () => void;
}) {
  return (
    <View style={styles.sectionHeaderRow}>
      <View style={{ flex: 1 }}>
        <Text style={styles.sectionTitle}>{title}</Text>
        {subtitle ? (
          <Text style={styles.sectionSubtitle}>{subtitle}</Text>
        ) : null}
      </View>
      {actionLabel ? (
        <Pressable onPress={onAction} hitSlop={8}>
          <Text style={styles.sectionAction}>{actionLabel}</Text>
        </Pressable>
      ) : null}
    </View>
  );
}

function BrandHeader() {
  return (
    <View style={styles.headerRow}>

      </Pressable>
    </View>
  );
}

function KpiCard({
  icon,
  label,
  value,
  delta,
  deltaPositive,
  accent,
}: {
  icon: keyof typeof Ionicons.glyphMap;
  label: string;
  value: string;
  delta?: string;
  deltaPositive?: boolean;
  accent: string;
}) {
  return (
    <View style={[styles.kpiCard, { width: SCREEN_W * 0.42 }]}>
      <View style={[styles.kpiIconWrap, { backgroundColor: accent + "22" }]}>
        <Ionicons name={icon} size={18} color={accent} />
      </View>
      <Text style={styles.kpiValue}>{value}</Text>
      <Text style={styles.kpiLabel}>{label}</Text>
      {delta ? (
        <View style={styles.kpiDeltaRow}>
          <Ionicons
            name={deltaPositive ? "trending-up" : "trending-down"}
            size={12}
            color={deltaPositive ? theme.success : theme.danger}
          />
          <Text
            style={[
              styles.kpiDeltaText,
              { color: deltaPositive ? theme.success : theme.danger },
            ]}
          >
            {delta}
          </Text>
        </View>
      ) : null}
    </View>
  );
}

function MetaFinancieraCard({ kpi }: { kpi: KpiFinanciero }) {
  const progreso = Math.min(kpi.ingresosMes / kpi.metaMes, 1);
  return (
    <View style={styles.metaCard}>
      <View style={styles.sectionHeaderRow}>
        <Text style={styles.metaTitle}>Meta del mes</Text>
        <Text style={styles.metaPct}>{Math.round(progreso * 100)}%</Text>
      </View>
      <View style={styles.progressTrack}>
        <View style={[styles.progressFill, { width: `${progreso * 100}%` }]} />
      </View>
      <View style={styles.metaFootRow}>
        <Text style={styles.metaFootText}>
          {fmtMoneda(kpi.ingresosMes)} de {fmtMoneda(kpi.metaMes)}
        </Text>
        <Text style={styles.metaFootMuted}>
          Faltan {fmtMoneda(Math.max(kpi.metaMes - kpi.ingresosMes, 0))}
        </Text>
      </View>
    </View>
  );
}

function PiezaTopCard({
  pieza,
  ranking,
}: {
  pieza: PiezaTopUI;
  ranking: number;
}) {
  return (
    <View style={styles.piezaCard}>
      <View style={styles.piezaImgWrap}>
        {pieza.imagenUrl ? (
          <Image source={{ uri: pieza.imagenUrl }} style={styles.piezaImg} />
        ) : (
          <Ionicons name="cube-outline" size={26} color={theme.textMuted} />
        )}
        <View style={styles.rankBadge}>
          <Text style={styles.rankBadgeText}>#{ranking}</Text>
        </View>
      </View>
      <Text style={styles.piezaNombre} numberOfLines={2}>
        {pieza.nombre}
      </Text>
      <Text style={styles.piezaMeta}>
        {pieza.vecesCotizada} cotizaciones · {pieza.ultimaVez}
      </Text>
      <Text style={styles.piezaPrecio}>
        {fmtMoneda(pieza.precioPromedio)} prom.
      </Text>
      <Pressable style={styles.reCotizarBtn}>
        <Ionicons name="refresh" size={13} color="#fff" />
        <Text style={styles.reCotizarText}>Cotizar de nuevo</Text>
      </Pressable>
    </View>
  );
}

function ImpresoraRow({ impresora }: { impresora: ImpresoraUI }) {
  const cfg = ESTADO_IMPRESORA_CFG[impresora.estado];
  const desgaste = Math.min(
    impresora.horasUsoTotal / impresora.vidaUtilHoras,
    1,
  );
  return (
    <View style={styles.impresoraRow}>
      <View style={[styles.estadoDot, { backgroundColor: cfg.color }]} />
      <View style={{ flex: 1 }}>
        <View style={styles.sectionHeaderRow}>
          <Text style={styles.impresoraModelo}>{impresora.modelo}</Text>
          <View style={[styles.estadoPill, { backgroundColor: cfg.bg }]}>
            <Ionicons name={cfg.icon} size={11} color={cfg.color} />
            <Text style={[styles.estadoPillText, { color: cfg.color }]}>
              {cfg.label}
            </Text>
          </View>
        </View>
        {impresora.pedidoActual ? (
          <Text style={styles.impresoraPedido} numberOfLines={1}>
            {impresora.pedidoActual}
          </Text>
        ) : (
          <Text style={styles.impresoraMuted}>{impresora.marca}</Text>
        )}
        <View style={styles.progressTrackSm}>
          <View
            style={[
              styles.progressFillSm,
              {
                width: `${desgaste * 100}%`,
                backgroundColor: desgaste > 0.85 ? theme.danger : theme.primary,
              },
            ]}
          />
        </View>
        <Text style={styles.impresoraDesgaste}>
          {impresora.horasUsoTotal.toLocaleString()}h /{" "}
          {impresora.vidaUtilHoras.toLocaleString()}h de vida útil
        </Text>
      </View>
    </View>
  );
}

function AlertaRow({ alerta }: { alerta: AlertaStockUI }) {
  const critico = alerta.nivel === "critico";
  return (
    <View
      style={[
        styles.alertaRow,
        { borderColor: critico ? theme.danger + "55" : theme.warning + "55" },
      ]}
    >
      <Ionicons
        name={
          alerta.tipo === "filamento"
            ? "color-filter-outline"
            : "layers-outline"
        }
        size={16}
        color={critico ? theme.danger : theme.warning}
      />
      <View style={{ flex: 1 }}>
        <Text style={styles.alertaNombre}>{alerta.nombre}</Text>
        <Text style={styles.alertaDetalle}>{alerta.detalle}</Text>
      </View>
    </View>
  );
}

function CotizacionPendienteRow({ cot }: { cot: CotizacionPendienteUI }) {
  const urgente = cot.diasEsperando >= 2;
  return (
    <View style={styles.cotRow}>
      <View style={{ flex: 1 }}>
        <Text style={styles.cotCodigo}>{cot.codigo}</Text>
        <Text style={styles.cotCliente} numberOfLines={1}>
          {cot.cliente}
        </Text>
      </View>
      <View style={{ alignItems: "flex-end" }}>
        <Text style={styles.cotPrecio}>{fmtMoneda(cot.precioFinal)}</Text>
        <Text style={[styles.cotDias, urgente && { color: theme.warning }]}>
          {cot.diasEsperando === 0 ? "Hoy" : `${cot.diasEsperando}d esperando`}
        </Text>
      </View>
    </View>
  );
}

function PedidoUrgenteChip({ pedido }: { pedido: PedidoUrgenteUI }) {
  const listo = pedido.estado === "listo";
  return (
    <View
      style={[
        styles.urgenteChip,
        { backgroundColor: listo ? theme.successBg : theme.warningBg },
      ]}
    >
      <Ionicons
        name={listo ? "checkmark-done" : "time-outline"}
        size={14}
        color={listo ? theme.success : theme.warning}
      />
      <View>
        <Text style={styles.urgenteCodigo}>
          {pedido.codigo} · {pedido.cliente}
        </Text>
        <Text
          style={[
            styles.urgenteTexto,
            { color: listo ? theme.success : theme.warning },
          ]}
        >
          {pedido.entregaTexto}
        </Text>
      </View>
    </View>
  );
}

            {MOCK_COTIZACIONES_PENDIENTES.map((c) => (
