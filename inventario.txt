const fmtMoneda = (n: number) => `Bs ${n.toLocaleString("es-BO")}`;
const costoPorGramo = (f: FilamentoUI) => f.costoCompra / f.capacidadRolloGramos;
const nivelStock = (f: FilamentoUI): "ok" | "bajo" | "critico" => {
  if (f.stockGramos <= f.umbralBajoStock * 0.5) return "critico";
  if (f.stockGramos <= f.umbralBajoStock) return "bajo";
  return "ok";
};

const ESTADO_IMPRESORA_CFG = {
  imprimiendo: { color: theme.success, bg: theme.successBg, label: "Imprimiendo", icon: "play-circle" as const },
  libre: { color: theme.info, bg: theme.infoBg, label: "Libre", icon: "checkmark-circle" as const },
  mantenimiento: { color: theme.warning, bg: theme.warningBg, label: "Mantenimiento", icon: "construct" as const },
};

const TABS: { key: TabInventario; label: string; icon: keyof typeof Ionicons.glyphMap }[] = [
  { key: "filamentos", label: "Filamentos", icon: "color-filter-outline" },
  { key: "piezas", label: "Piezas en stock", icon: "layers-outline" },
  { key: "impresoras", label: "Impresoras", icon: "print-outline" },
];

function KpiChip({ icon, label, value, color }: { icon: keyof typeof Ionicons.glyphMap; label: string; value: string; color: string }) {
  return (
    <View style={styles.kpiChip}>
      <View style={[styles.kpiIconWrap, { backgroundColor: color + "22" }]}>
        <Ionicons name={icon} size={16} color={color} />
      </View>
      <View>
        <Text style={styles.kpiValue}>{value}</Text>
        <Text style={styles.kpiLabel}>{label}</Text>
      </View>
    </View>
  );
}

function PiezaTopChip({ pieza }: { pieza: PiezaTopUI }) {
  return (
    <View style={styles.piezaTopChip}>
      <View style={{ flex: 1 }}>
        <Text style={styles.piezaTopNombre} numberOfLines={1}>{pieza.nombre}</Text>
        <Text style={styles.piezaTopMeta}>{pieza.vecesCotizada}x · {fmtMoneda(pieza.precioPromedio)}</Text>
      </View>
      <Pressable style={styles.reCotizarBtn}>
        <Ionicons name="refresh" size={13} color="#fff" />
      </Pressable>
    </View>
  );
}

function ImagePlaceholder({ uri, icon, size = 90 }: { uri?: string; icon: keyof typeof Ionicons.glyphMap; size?: number }) {
  return (
    <View style={[styles.imgWrap, { height: size }]}>
      {uri ? <Image source={{ uri }} style={styles.img} /> : <Ionicons name={icon} size={28} color={theme.textMuted} />}
    </View>
  );
}

function FilamentoCard({ f, onPress }: { f: FilamentoUI; onPress: () => void }) {
  const nivel = nivelStock(f);
  const pct = Math.min(f.stockGramos / f.capacidadRolloGramos, 1);
  const nivelColor = nivel === "critico" ? theme.danger : nivel === "bajo" ? theme.warning : theme.success;

  return (
    <Pressable style={styles.gridCard} onPress={onPress}>
      <ImagePlaceholder uri={f.imagenUrl} icon="color-filter-outline" />
      <View style={styles.colorRow}>
        <View style={[styles.colorDot, { backgroundColor: f.colorHex }]} />
        <Text style={styles.gridTitle} numberOfLines={1}>{f.material} · {f.color}</Text>
      </View>
      <Text style={styles.gridSubtitle}>{f.marca}</Text>

      <View style={styles.progressTrackSm}>
        <View style={[styles.progressFillSm, { width: `${pct * 100}%`, backgroundColor: nivelColor }]} />
      </View>
      <Text style={[styles.gridMeta, nivel !== "ok" && { color: nivelColor, fontWeight: "700" }]}>
        {f.stockGramos}g / {f.capacidadRolloGramos}g {nivel !== "ok" ? "· Stock bajo" : ""}
      </Text>
      <Text style={styles.gridPrecio}>{costoPorGramo(f).toFixed(3)} Bs/g</Text>
    </Pressable>
  );
}

function PiezaCard({ p }: { p: PiezaStockUI }) {
  return (
    <View style={styles.gridCard}>
      <ImagePlaceholder icon="cube-outline" />
      <Text style={styles.gridTitle} numberOfLines={2}>{p.nombre}</Text>
      <Text style={styles.gridSubtitle}>Cantidad: {p.cantidad}</Text>
      <View
        style={[
          styles.asignadaPill,
          { backgroundColor: p.asignada ? theme.infoBg : theme.successBg },
        ]}
      >
        <Text style={[styles.asignadaPillText, { color: p.asignada ? theme.info : theme.success }]}>
          {p.asignada ? `Reservada · ${p.cliente}` : "Sin asignar"}
        </Text>
      </View>
      <Text style={styles.gridMeta}>{p.fechaImpresion}</Text>
      <Text style={styles.gridPrecio}>{fmtMoneda(p.precioVenta)}</Text>
    </View>
  );
}

function ImpresoraCard({ imp, onPress }: { imp: ImpresoraUI; onPress: () => void }) {
  const cfg = ESTADO_IMPRESORA_CFG[imp.estado];
  const desgaste = Math.min(imp.horasUsoTotal / imp.vidaUtilHoras, 1);

  return (
    <Pressable style={styles.gridCard} onPress={onPress}>
      <ImagePlaceholder uri={imp.imagenUrl} icon="print-outline" />
      <Text style={styles.gridTitle} numberOfLines={1}>{imp.modelo}</Text>
      <Text style={styles.gridSubtitle}>{imp.marca}</Text>

      <View style={[styles.estadoPill, { backgroundColor: cfg.bg, alignSelf: "flex-start" }]}>
        <Ionicons name={cfg.icon} size={11} color={cfg.color} />
        <Text style={[styles.estadoPillText, { color: cfg.color }]}>{cfg.label}</Text>
      </View>

      {imp.pedidoActual && (
        <Text style={styles.gridMeta} numberOfLines={1}>{imp.pedidoActual}</Text>
      )}

      <View style={styles.progressTrackSm}>
        <View
          style={[
            styles.progressFillSm,
            { width: `${desgaste * 100}%`, backgroundColor: desgaste > 0.85 ? theme.danger : theme.primary },
          ]}
        />
      </View>
      <Text style={styles.gridMeta}>{imp.horasUsoTotal.toLocaleString()}h / {imp.vidaUtilHoras.toLocaleString()}h</Text>
    </Pressable>
  );
}

function DetalleModal({
  visible,
  onClose,
  title,
  imagenUrl,
  icon,
  rows,
}: {
  visible: boolean;
  onClose: () => void;
  title: string;
  imagenUrl?: string;
  icon: keyof typeof Ionicons.glyphMap;
  rows: { label: string; value: string; color?: string }[];
}) {
  return (
    <Modal visible={visible} animationType="slide" transparent onRequestClose={onClose}>
      <View style={styles.modalBackdrop}>
        <View style={styles.modalSheet}>
          <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={{ paddingBottom: 24 }}>
            <View style={styles.modalHandle} />
            <View style={styles.sectionHeaderRow}>
              <Text style={styles.modalTitle}>{title}</Text>
              <Pressable onPress={onClose} hitSlop={10}>
                <Ionicons name="close" size={22} color={theme.textMuted} />
              </Pressable>
            </View>
            <ImagePlaceholder uri={imagenUrl} icon={icon} size={140} />
            <View style={{ marginTop: 12, gap: 10 }}>
              {rows.map((r) => (
                <View key={r.label} style={styles.detailRow}>
                  <Text style={styles.detailLabel}>{r.label}</Text>
                  <Text style={[styles.detailValue, r.color && { color: r.color }]}>{r.value}</Text>
                </View>
              ))}
            </View>
          </ScrollView>
        </View>
      </View>
    </Modal>
  );
}

export default function InventarioScreen() {
  const [tab, setTab] = useState<TabInventario>("filamentos");
  const [busqueda, setBusqueda] = useState("");
  const [filamentoActivo, setFilamentoActivo] = useState<FilamentoUI | null>(null);
  const [impresoraActiva, setImpresoraActiva] = useState<ImpresoraUI | null>(null);

  const kpis = useMemo(() => {
    const valorFilamentos = MOCK_FILAMENTOS.reduce((acc, f) => acc + costoPorGramo(f) * f.stockGramos, 0);
    const valorImpresoras = MOCK_IMPRESORAS.reduce((acc, i) => acc + i.costoCompra, 0);
    const bajoStock = MOCK_FILAMENTOS.filter((f) => nivelStock(f) !== "ok").length;
    const piezasSinAsignar = MOCK_PIEZAS.filter((p) => !p.asignada).reduce((acc, p) => acc + p.cantidad, 0);
    return { valorTotal: valorFilamentos + valorImpresoras, bajoStock, piezasSinAsignar };
  }, []);

  const filamentosFiltrados = useMemo(
    () => MOCK_FILAMENTOS.filter((f) => `${f.marca} ${f.material} ${f.color}`.toLowerCase().includes(busqueda.toLowerCase())),
    [busqueda]
  );
  const piezasFiltradas = useMemo(
    () => MOCK_PIEZAS.filter((p) => p.nombre.toLowerCase().includes(busqueda.toLowerCase())),
    [busqueda]
  );
  const impresorasFiltradas = useMemo(
    () => MOCK_IMPRESORAS.filter((i) => `${i.marca} ${i.modelo}`.toLowerCase().includes(busqueda.toLowerCase())),
    [busqueda]
  );

  const dataActual = tab === "filamentos" ? filamentosFiltrados : tab === "piezas" ? piezasFiltradas : impresorasFiltradas;

  return (
    <SafeAreaView style={styles.safe} edges={["top"]}>
      <FlatList
        key={tab}
        data={dataActual as any[]}
        numColumns={2}
        keyExtractor={(item) => item.id}
        columnWrapperStyle={styles.columnWrapper}
        contentContainerStyle={styles.listContent}
        showsVerticalScrollIndicator={false}
        renderItem={({ item }) => {
          if (tab === "filamentos") return <FilamentoCard f={item as FilamentoUI} onPress={() => setFilamentoActivo(item as FilamentoUI)} />;
          if (tab === "piezas") return <PiezaCard p={item as PiezaStockUI} />;
          return <ImpresoraCard imp={item as ImpresoraUI} onPress={() => setImpresoraActiva(item as ImpresoraUI)} />;
        }}
        ListEmptyComponent={
          <View style={styles.emptyState}>
            <Ionicons name="file-tray-outline" size={30} color={theme.textMuted} />
            <Text style={styles.emptyTitle}>Sin resultados para esta búsqueda</Text>
          </View>
        }
        ListHeaderComponent={
          <View style={{ gap: 18, marginBottom: 6 }}>
            <Text style={styles.screenTitle}>Inventario</Text>

            {/* KPIs */}
            <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={{ gap: 10, paddingRight: 16 }}>
              <KpiChip icon="wallet-outline" label="Valor total" value={fmtMoneda(Math.round(kpis.valorTotal))} color={theme.primary} />
              <KpiChip icon="alert-circle-outline" label="Bajo stock" value={`${kpis.bajoStock}`} color={theme.danger} />
              <KpiChip icon="layers-outline" label="Piezas sin asignar" value={`${kpis.piezasSinAsignar}`} color={theme.info} />
            </ScrollView>

            {/* Piezas más cotizadas */}
            <View>
              <View style={styles.sectionHeaderRow}>
                <Text style={styles.sectionTitle}>Piezas más cotizadas</Text>
                <Text style={styles.sectionAction}>Ver todas</Text>
              </View>
              <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={{ gap: 10, paddingRight: 16 }}>
                {MOCK_PIEZAS_TOP.map((p) => (
                  <PiezaTopChip key={p.id} pieza={p} />
                ))}
              </ScrollView>
            </View>

            {/* Tabs */}
            <View style={styles.tabsRow}>
              {TABS.map((t) => {
                const activo = tab === t.key;
                return (
                  <Pressable
                    key={t.key}
                    onPress={() => {
                      setTab(t.key);
                      setBusqueda("");
                    }}
                    style={[styles.tabBtn, activo && styles.tabBtnActivo]}
                  >
                    <Ionicons name={t.icon} size={15} color={activo ? "#fff" : theme.textMuted} />
                    <Text style={[styles.tabBtnText, activo && { color: "#fff" }]}>{t.label}</Text>
                  </Pressable>
                );
              })}
            </View>

            {/* Buscador + agregar */}
            <View style={{ flexDirection: "row", gap: 10 }}>
              <View style={[styles.searchBox, { flex: 1 }]}>
                <Ionicons name="search" size={16} color={theme.textMuted} />
                <TextInput
                  value={busqueda}
                  onChangeText={setBusqueda}
                  placeholder={`Buscar en ${TABS.find((t) => t.key === tab)?.label.toLowerCase()}`}
                  placeholderTextColor={theme.textMuted}
                  style={styles.searchInput}
                />
              </View>
              {tab !== "piezas" && (
                <Pressable style={styles.addBtn}>
                  <Ionicons name="add" size={20} color="#fff" />
                </Pressable>
              )}
            </View>
          </View>
        }
      />

      <DetalleModal
        visible={!!filamentoActivo}
        onClose={() => setFilamentoActivo(null)}
        title={filamentoActivo ? `${filamentoActivo.material} · ${filamentoActivo.color}` : ""}
        icon="color-filter-outline"
        imagenUrl={filamentoActivo?.imagenUrl}
        rows={
          filamentoActivo
            ? [
                { label: "Marca", value: filamentoActivo.marca },
                { label: "Stock actual", value: `${filamentoActivo.stockGramos}g de ${filamentoActivo.capacidadRolloGramos}g` },
                { label: "Costo por gramo", value: `${costoPorGramo(filamentoActivo).toFixed(3)} Bs/g` },
                { label: "Costo del rollo", value: fmtMoneda(filamentoActivo.costoCompra) },
                {
                  label: "Estado de stock",
                  value: nivelStock(filamentoActivo) === "ok" ? "Normal" : nivelStock(filamentoActivo) === "bajo" ? "Bajo" : "Crítico",
                  color: nivelStock(filamentoActivo) === "ok" ? theme.success : nivelStock(filamentoActivo) === "bajo" ? theme.warning : theme.danger,
                },
              ]
            : []
        }
      />

      <DetalleModal
        visible={!!impresoraActiva}
        onClose={() => setImpresoraActiva(null)}
        title={impresoraActiva?.modelo ?? ""}
        icon="print-outline"
        imagenUrl={impresoraActiva?.imagenUrl}
        rows={
          impresoraActiva
            ? [
                { label: "Marca", value: impresoraActiva.marca },
                { label: "Estado", value: ESTADO_IMPRESORA_CFG[impresoraActiva.estado].label, color: ESTADO_IMPRESORA_CFG[impresoraActiva.estado].color },
                { label: "Pedido actual", value: impresoraActiva.pedidoActual ?? "Ninguno" },
                { label: "Horas de uso", value: `${impresoraActiva.horasUsoTotal.toLocaleString()}h / ${impresoraActiva.vidaUtilHoras.toLocaleString()}h` },
                { label: "Consumo", value: `${impresoraActiva.potenciaWatts}W` },
                { label: "Costo de compra", value: fmtMoneda(impresoraActiva.costoCompra) },
              ]
            : []
        }
      />
    </SafeAreaView>
  );
}
